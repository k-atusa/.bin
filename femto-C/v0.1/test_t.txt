// std 모듈 포함 (println 등)
include std;

// Node 구조체 정의
struct Node {
    value: int;
    prev: ?box Node; // 이전 노드 (nullable 소유권)
    next: ?box Node; // 다음 노드 (nullable 소유권)
}

// Node 생성자
func Node.__new(value: int) : box Node {
    var new_node: box Node = new Node;
    new_node.value = value;
    new_node.prev = void; // ?box T는 void로 초기화 가능
    new_node.next = void;
    return new_node;
}

// LinkedList 구조체 정의
struct LinkedList {
    head: ?box Node; // 리스트의 첫 노드
    tail: ?box Node; // 리스트의 마지막 노드 (옵션: 빠른 tail 접근)
    count: uint;
}

// LinkedList 생성자
func LinkedList.__new() : box LinkedList {
    var list: box LinkedList = new LinkedList;
    list.head = void;
    list.tail = void;
    list.count = 0;
    return list;
}

// 리스트의 맨 앞에 노드 추가 (push_front)
func (self: &mut LinkedList) push_front(value: int) {
    var new_node: box Node = Node.__new(value);

    if (self.head != void) { // 기존 헤드가 있다면
        // 기존 헤드 노드의 prev를 새 노드로 연결
        var old_head: &mut Node = &(self.head as box Node); // ?box -> box -> &mut
        old_head.prev = new_node; // new_node의 소유권은 여전히 new_node에 있음. old_head.prev는 새 노드 참조
                                  // NOTE: old_head.prev = new_node;는 box Node로의 할당이므로
                                  // new_node의 소유권이 이동하거나, new_node가 복사되어야 함.
                                  // 여기서는 new_node에 대한 '약한 참조'로 동작해야 하므로
                                  // ?&Node 같은 타입이 더 적합할 수 있음.
                                  // 현 문법상으로는 old_head.prev가 new_node의 복사본을 받거나,
                                  // new_node의 소유권을 가져가버릴 수 있어 문제가 발생.
                                  // ====> 'prev: ?box Node'가 아니라 'prev: ?&Node'가 더 자연스럽다.
                                  // (이를 반영하여 아래 코드 수정)
    }

    // 새 노드의 소유권을 리스트의 head로 이동
    new_node.next = self.head; // 기존 head의 소유권이 new_node.next로 이동
    self.head = new_node;      // new_node의 소유권이 self.head로 이동

    if (self.tail == void) { // 리스트가 비어있었다면
        self.tail = &self.head; // head가 첫 노드이므로 tail도 이 노드를 참조
                               // NOTE: 여기서 tail은 '소유권'이 아닌 '참조'이므로
                               // ?box Node가 아니라 ?&Node나 ?&mut Node여야 한다.
                               // (이를 반영하여 아래 코드 수정)
    }
    self.count = self.count + 1;
    std.println("Pushed front: ", value);
}

// ======= 수정된 LinkedList 구조체 정의 (더 현실적인 prev/tail 타입 반영) ========
struct LinkedList {
    head: ?box Node;     // 리스트의 첫 노드 (소유권)
    tail: ?&mut Node;    // 리스트의 마지막 노드에 대한 가변 참조 (빌림)
    count: uint;
}

// 수정된 Node 구조체 정의 (이중 연결 리스트의 Prev는 보통 소유권이 아님)
struct Node {
    value: int;
    // prev: ?box Node;  // prev는 소유권이 아니므로 ?box T는 적절치 않음
    prev: ?&mut Node; // 이전 노드에 대한 가변 참조 (빌림)
    next: ?box Node;  // 다음 노드에 대한 소유권
}

// 수정된 Node 생성자
func Node.__new(value: int) : box Node {
    var new_node: box Node = new Node;
    new_node.value = value;
    new_node.prev = void; // ?&mut T는 void로 초기화 가능
    new_node.next = void;
    return new_node;
}

// 수정된 LinkedList 생성자
func LinkedList.__new() : box LinkedList {
    var list: box LinkedList = new LinkedList;
    list.head = void;
    list.tail = void;
    list.count = 0;
    return list;
}

// 수정된 push_front 메서드
func (self: &mut LinkedList) push_front(value: int) {
    var new_node: box Node = Node.__new(value);

    // 기존 헤드가 있다면, 기존 헤드의 prev를 새 노드로 연결
    if (self.head != void) {
        var old_head_mut_ref: &mut Node = &(self.head as box Node); // ?box -> box -> &mut
        old_head_mut_ref.prev = &mut new_node; // new_node에 대한 가변 참조를 old_head_mut_ref.prev에 저장
    }

    // 새 노드의 next를 기존 head로 설정 (소유권 이동)
    new_node.next = self.head; // 기존 head의 소유권이 new_node.next로 이동

    // 새 노드의 소유권을 리스트의 head로 설정
    self.head = new_node;      // new_node의 소유권이 self.head로 이동

    if (self.tail == void) { // 리스트가 비어있었다면
        self.tail = &(self.head as box Node); // head 노드에 대한 가변 참조를 tail에 저장
    }
    self.count = self.count + 1;
    std.println("Pushed front: ", value);
}

// 리스트의 맨 뒤에 노드 추가 (push_back)
func (self: &mut LinkedList) push_back(value: int) {
    var new_node: box Node = Node.__new(value);

    if (self.tail != void) { // 기존 꼬리가 있다면
        var old_tail_mut_ref: &mut Node = &(self.tail as &mut Node); // ?&mut -> &mut
        old_tail_mut_ref.next = new_node; // old_tail_mut_ref.next가 new_node의 소유권을 가져감
    } else { // 리스트가 비어있다면
        self.head = new_node; // 새 노드가 헤드가 됨
    }
    self.tail = &mut new_node; // tail은 항상 마지막 노드를 참조 (new_node는 이미 소유권이 이동했으므로, 이 참조는 위험할 수 있음)
                               // 이 부분은 Rust에서 self.tail = Some(new_node); 처럼 박싱된 값에 대한 참조를 얻는 것과 유사.
                               // self.tail = &mut (self.head as box Node); 가 더 안전할 수 있다.
                               // (이를 반영하여 아래 코드 수정)

    self.count = self.count + 1;
    std.println("Pushed back: ", value);
}

// 수정된 push_back (tail을 안전하게 업데이트하기 위해 head부터 순회하는 방식)
// 또는, tail이 ?&mut Node이므로, tail을 참조하는 노드의 next 필드를 직접 수정할 수 있음.
func (self: &mut LinkedList) push_back_refined(value: int) {
    var new_node: box Node = Node.__new(value);

    if (self.tail != void) { // 기존 꼬리가 있다면
        var old_tail_mut_ref: &mut Node = &(self.tail as &mut Node);
        // old_tail_mut_ref의 next에 새 노드의 소유권을 넘김
        old_tail_mut_ref.next = new_node;
        // 이제 새 노드는 old_tail_mut_ref.next에 소유되어 있음
        // 새 노드의 prev를 이전 꼬리로 설정
        (old_tail_mut_ref.next as box Node).prev = old_tail_mut_ref; // 여기서 as box Node는 위험할 수 있음.
                                                                    // old_tail_mut_ref.next는 ?box Node이므로,
                                                                    // 유효한지 확인하고 box Node로 캐스팅해야 함.
                                                                    // 또한 (old_tail_mut_ref.next as box Node) 자체가 새로운 box를 만들 수도 있음.
                                                                    // ===> 여기서 &mut (old_tail_mut_ref.next as box Node) 가 필요.
        self.tail = &mut (old_tail_mut_ref.next as box Node); // 새로운 tail로 업데이트
    } else { // 리스트가 비어있다면
        self.head = new_node; // 새 노드가 head
        self.tail = &(self.head as box Node); // 새 노드가 tail
    }
    self.count = self.count + 1;
    std.println("Pushed back: ", value);
}


// 리스트의 모든 값을 출력
func (self: &LinkedList) print_list() {
    std.print("List (count: ", self.count, "): [");
    var current: ?&Node = void; // ?&Node를 사용하여 널 가능성 처리

    if (self.head != void) {
        current = &(self.head as box Node); // ?box -> box -> &
    }

    while (current != void) {
        var node_ref: &Node = (current as &Node); // ?&Node -> &Node
        std.print(node_ref.value);
        if (node_ref.next != void) { // 다음 노드가 있다면
            std.print(" <-> ");
            current = &(node_ref.next as box Node); // ?box -> box -> &
        } else {
            current = void;
        }
    }
    std.println("]");
}

include c.stdio; // malloc, free 대신 c.stdio.printf 사용 예시

// 스택의 노드 (데이터는 *void로 저장)
struct StackNode {
    data: *void; // 모든 타입의 데이터를 가리킬 수 있는 로우 포인터
    next: ?box StackNode; // 다음 노드에 대한 소유권 (nullable)
}

// Stack 구조체
struct Stack {
    top: ?box StackNode; // 스택의 최상단 노드
    count: uint;
}

// Stack 생성자
func Stack.__new() : box Stack {
    var stack: box Stack = new Stack;
    stack.top = void;
    stack.count = 0;
    return stack;
}

// 스택에 데이터 push (box T를 받아 *void로 저장)
func (self: &mut Stack) push(data_box: box void) { // box void는 모든 box T를 받을 수 있다고 가정 (아니면 box T 타입의 제네릭 필요)
                                                 // NOTE: 현재 문법에서 box void는 모든 box T를 받을 수 있다고 정의되지 않았음.
                                                 // 이 경우 void*처럼 임의의 데이터를 저장하기 위해선 raw pointer를 사용해야 함.
                                                 // 임시방편으로 'box void'를 '모든 box 타입의 캡슐화'로 해석.
                                                 // 더 정확한 구현은 'box T' 타입 제네릭이 필요함.

    var new_node: box StackNode = new StackNode;
    new_node.data = data_box as *void; // box T의 로우 포인터를 *void로 저장. 소유권은 여전히 data_box에 있음.
                                       // 하지만 StackNode가 data를 '참조'할 뿐 '소유'하지 않으므로
                                       // data_box는 push 후에도 유효하게 유지되어야 함.
                                       // 이는 Stack이 data_box의 소유권을 가져와야 함을 의미.
                                       // ===> data: box void; 가 더 적절하다. (아래 코드 수정)

    // ======= 수정된 StackNode 구조체 (데이터 소유권을 StackNode가 가짐) ========
    struct StackNode {
        data: box void; // 모든 box T의 소유권을 가질 수 있는 필드 (가정)
        next: ?box StackNode;
    }

    // 수정된 push
    func (self: &mut Stack) push(data_box: box void) {
        var new_node: box StackNode = new StackNode;
        new_node.data = data_box; // data_box의 소유권이 new_node.data로 이동!

        // 기존 top의 소유권을 새 노드의 next로 이동
        new_node.next = self.top;
        self.top = new_node;
        self.count = self.count + 1;
        std.println("Pushed an item.");
    }

    // 스택에서 데이터 pop (unsafe 사용)
    func (self: &mut Stack) !pop() : ?box void { // !unsafe_function으로 선언
        if (self.top == void) {
            std.println("Stack is empty.");
            return void;
        }

        var old_top: box StackNode = (self.top as box StackNode); // ?box -> box
        // 새로운 top은 이전 top의 next 노드가 됨 (소유권 이동)
        self.top = old_top.next;

        self.count = self.count - 1;
        std.println("Popped an item.");

        // old_top.data의 소유권을 호출자에게 돌려줌
        return old_top.data; // box void 반환
    }

    func main() {
        std.println("--- Femto C Type-unsafe Stack Example ---");

        var my_stack: box Stack = Stack.__new();

        // 정수와 문자열 박싱하여 스택에 push
        my_stack.push(new int = 123 as box void); // int를 box로 만들어 push (box T -> box void)
        my_stack.push(new [6]u8 = {'h','e','l','l','o','\0'} as box void); // 문자열을 box로 만들어 push

        my_stack.push(new int = 456 as box void);
        my_stack.push(new [6]u8 = {'w','o','r','l','d','\0'} as box void);

        // pop하여 사용 (unsafe)
        var item1_void: ?box void = !my_stack.pop();
        if (item1_void != void) {
            unsafe { // *T 역참조는 unsafe
                var item1_str: box []u8 = (item1_void as box []u8); // box void -> box []u8 (unsafe)
                c.stdio.printf(&"Popped string: %s\n\0"[0] as *char, &item1_str[0] as *char);
                // item1_str은 스코프 벗어날 때 자동 해제
            }
        }

        var item2_void: ?box void = !my_stack.pop();
        if (item2_void != void) {
            unsafe {
                var item2_int: box int = (item2_void as box int); // box void -> box int (unsafe)
                std.println("Popped integer: ", *item2_int);
                // item2_int는 스코프 벗어날 때 자동 해제
            }
        }

        !my_stack.pop(); // hello
        !my_stack.pop(); // 123
        !my_stack.pop(); // empty

        std.println("--- End of Example ---");
    }

include c.stdio; // C 표준 입출력 라이브러리
include c.stdlib; // C 표준 라이브러리 (malloc, free)
include c.string; // C 문자열 함수 (strlen, strcpy)

// C 함수 선언 (c.stdio.printf, c.stdlib.malloc, c.stdlib.free, c.string.strlen, c.string.strcpy)
// 이 함수들은 컴파일러가 알아서 C 라이브러리와 링크되도록 처리한다고 가정.

// Femto C에서 C 스타일 문자열을 받아서 처리하는 함수
// C 문자열은 *char (로우 포인터) 이므로 안전하지 않은 작업이 필요
func !process_c_string(c_str: *char) { // !unsafe_function
    unsafe { // 로우 포인터 역참조 및 산술 연산은 unsafe
        var len: int = c.string.strlen(c_str) as int;
        c.stdio.printf(&"Received C string: %s (Length: %d)\n\0"[0] as *char, c_str, len);

        // C 문자열을 Femto C의 box []u8로 복사하여 소유권 가져오기
        var femto_str_box: box []u8 = new [len + 1]u8; // +1 for null terminator
        // *char -> *u8 캐스팅은 안전함
        c.string.strcpy(femto_str_box[0] as *char, c_str); // Femto C 배열의 첫 주소
        
        std.println("Copied to Femto C box string: ", &femto_str_box); // &box []u8는 &[]u8로 자동 변환
        // femto_str_box는 이 함수 스코프를 벗어날 때 자동으로 해제됨.
    }
}

// Femto C box []u8를 C 함수에 전달하는 함수
func call_c_function_with_femto_string(femto_str: box []u8) {
    unsafe { // C 함수 호출에 로우 포인터를 넘기므로 unsafe 필요
        // Femto C의 box []u8에서 로우 포인터 *char를 얻음
        // box []u8 -> *u8 -> *char (unsafe)
        // femto_str[0]의 주소는 &femto_str[0] as *char 로 가능
        var c_str_ptr: *char = &femto_str[0] as *char; // 로우 포인터 연동

        c.stdio.printf(&"Calling C function with Femto C string: %s\n\0"[0] as *char, c_str_ptr);
    }
    // 함수가 끝나면 femto_str의 소유권이 이 함수에 있었으므로 자동 해제됨.
}

// C 라이브러리에서 메모리를 할당받아 Femto C box로 가져오는 함수
func !allocate_from_c_and_wrap(size: uint) : ?box []u8 { // !unsafe_function
    unsafe { // malloc, *T -> box T는 unsafe
        var c_raw_ptr: *void = c.stdlib.malloc(size as uint);
        if (c_raw_ptr == void) {
            return void; // 메모리 할당 실패
        }

        // C에서 할당된 메모리의 소유권을 Femto C의 box []u8로 이동
        // *void -> *u8 -> box []u8 (unsafe)
        var wrapped_box: box []u8 = (c_raw_ptr as *u8 as box []u8);
        return wrapped_box;
    }
}

func main() {
    std.println("--- Femto C FFI Example ---");

    // 1. C 문자열을 Femto C로 가져와 처리
    unsafe { // C 문자열 리터럴 접근 및 포인터 캐스팅
        !process_c_string(&"Hello from C!\0"[0] as *char);
    }

    // 2. Femto C 문자열을 C 함수에 전달
    var my_femto_str: box []u8 = new [10]u8 = {'F','e','m','t','o','C','!','\0','\0','\0'};
    call_c_function_with_femto_string(my_femto_str); // my_femto_str 소유권이 함수로 이동 후 해제됨

    // 3. C에서 할당된 메모리를 Femto C box로 래핑
    var c_allocated_data: ?box []u8 = !allocate_from_c_and_wrap(16); // 16바이트 할당
    if (c_allocated_data != void) {
        std.println("Successfully wrapped C allocated memory.");
        // c_allocated_data는 스코프 벗어날 때 자동으로 c.stdlib.free로 해제됨 (box []u8의 __delete가 malloc된 메모리를 free하도록 정의된다고 가정)
    } else {
        std.println("Failed to allocate memory from C.");
    }

    std.println("--- End of Example ---");
}

struct FileError {
    code: int;
    path: box []u8; // 오류가 발생한 파일 경로
    message: box []u8; // 상세 오류 메시지

// FileError 생성자 (__new)
func FileError.__new(code: int, path_str: char[], msg_str: char[]) : box FileError {
    var error_obj: box FileError = new FileError;
    error_obj.code = code;
    // 문자열 복사 (std 라이브러리에 from_cstr 같은 함수가 있다고 가정)
    error_obj.path = std.string.from_cstr(path_str);
    error_obj.message = std.string.from_cstr(msg_str);
    return error_obj;
}

// 2. 다른 타입의 사용자 정의 예외
struct NetworkError {
    code: int;
    host: box []u8;
    port: int;
}

// NetworkError 생성자 (__new)
func NetworkError.__new(code: int, host_str: char[], port_num: int) : box NetworkError {
    var error_obj: box NetworkError = new NetworkError;
    error_obj.code = code;
    error_obj.host = std.string.from_cstr(host_str);
    error_obj.port = port_num;
    return error_obj;
}

// 파일을 처리하는 가상 함수 (예외를 발생시킬 수 있음)
func process_file(file_name: char[], content: char[], fail_scenario: int) {
    std.println("Processing file: ", &file_name); // char[]는 &[]u8로 변환 가능하다고 가정

    if (fail_scenario == 1) {
        raise FileError.__new(101, file_name, "File not found or access denied.");
    } else if (fail_scenario == 2) {
        raise FileError.__new(102, file_name, "Disk full during write operation.");
    } else if (fail_scenario == 3) {
        raise NetworkError.__new(201, "example.com", 8080);
    }
    std.println("File processed successfully: ", &file_name);
}

func main() {
    std.println("--- Femto C Exception Handling Example ---");

    // 1. 파일 처리 시나리오 1: 파일 없음 (FileError)
    try {
        std.println("\nAttempting file operation 1 (FileError expected)...");
        process_file("report.txt", "some data", 1);
        std.println("This line will not be executed if exception is raised.");
    } catch (var e: box FileError) { // FileError 타입만 catch
        std.println("Caught FileError!");
        std.println("Code: ", e.code);
        std.println("Path: ", &e.path);
        std.println("Message: ", &e.message);
        // e (box FileError)는 스코프 종료 시 자동 해제
    } catch (var e: box NetworkError) { // NetworkError는 여기서 잡히지 않음
        std.println("Caught NetworkError (unexpected in this block)!");
        // 이 catch는 FileError가 발생하면 도달하지 않음.
    }

    // 2. 파일 처리 시나리오 2: 네트워크 오류 (NetworkError)
    // 이 try-catch는 NetworkError를 잡을 수 있지만 FileError는 못 잡음
    try {
        std.println("\nAttempting file operation 2 (NetworkError expected)...");
        process_file("config.json", "network settings", 3);
        std.println("This line will not be executed if exception is raised.");
    } catch (var e: box NetworkError) { // NetworkError 타입만 catch
        std.println("Caught NetworkError!");
        std.println("Code: ", e.code);
        std.println("Host: ", &e.host);
        std.println("Port: ", e.port);
        // e (box NetworkError)는 스코프 종료 시 자동 해제
    } catch (var e: box FileError) {
        std.println("Caught FileError (unexpected in this block)!");
    }

    // 3. 파일 처리 시나리오 3: 성공
    try {
        std.println("\nAttempting file operation 3 (Success expected)...");
        process_file("log.csv", "new entry", 0);
        std.println("File operation 3 completed successfully.");
    } catch (var e: box FileError) {
        std.println("Caught FileError (should not happen here)!");
    } catch (var e: box NetworkError) {
        std.println("Caught NetworkError (should not happen here)!");
    }

    std.println("\n--- End of Example ---");
}

include std;
include c.stdio; // fopen, fclose, fprintf, printf
include c.stdlib; // malloc, free

// 가상의 파일 핸들 타입 (로우 포인터)
type FileHandle = *void;

// 가상의 파일 열기 함수 (C 라이브러리 함수 호출 가정)
// 파일 열기는 unsafe가 필요 (로우 포인터 반환, C 함수 호출)
func !open_file(path: *char, mode: *char) : ?FileHandle {
    unsafe {
        var handle: FileHandle = c.stdio.fopen(path, mode);
        if (handle == void) {
            std.println("Error: Failed to open file!");
        }
        return handle;
    }
}

// 가상의 파일 닫기 함수 (C 라이브러리 함수 호출 가정)
// 파일 닫기는 unsafe가 필요 (로우 포인터 조작)
func !close_file(handle: FileHandle) {
    unsafe {
        c.stdio.fclose(handle);
        std.println("File closed.");
    }
}

// 파일을 처리하고 (선택적으로) 에러를 발생시키는 함수
func process_data_in_file(file_name: char[], fail_scenario: int) {
    std.println("Attempting to process data in file: ", &file_name);

    // 파일 열기
    var file_handle: ?FileHandle = void;
    unsafe {
        file_handle = !open_file(&file_name[0] as *char, &"w\0"[0] as *char); // 파일 쓰기 모드
    }

    if (file_handle == void) {
        std.println("File processing aborted due to open failure.");
        return; // 파일 열기 실패 시 즉시 종료
    }

    // defer를 사용하여 함수 종료 시 파일 핸들을 자동으로 닫도록 예약
    // file_handle은 함수 종료 시에도 유효해야 함.
    defer !close_file(file_handle as FileHandle); // !close_file 호출 시 file_handle은 유효할 것임.
                                               // ?FileHandle을 FileHandle로 캐스팅해야 할 수 있음.

    // 데이터 쓰기 (예시)
    unsafe {
        c.stdio.fprintf(file_handle as *c.stdio._IO_FILE, &"Data Line 1\n\0"[0] as *char);
        c.stdio.fprintf(file_handle as *c.stdio._IO_FILE, &"Data Line 2\n\0"[0] as *char);
    }
    std.println("Data written to file.");

    // 특정 시나리오에서 강제로 예외 발생
    if (fail_scenario == 1) {
        std.println("Simulating an error during file processing...");
        raise std.err.__new(301, &"file.txt"[0] as *char, "Simulated processing error.");
    }

    std.println("File processing completed successfully.");
}

// malloc/free와 defer 조합 예시
func !allocate_and_cleanup_with_defer(size: uint) : *void {
    unsafe {
        var raw_ptr: *void = c.stdlib.malloc(size);
        if (raw_ptr == void) {
            std.println("Memory allocation failed!");
            return void;
        }
        defer c.stdlib.free(raw_ptr); // 함수 종료 시 raw_ptr 메모리 자동 해제
        std.println("Memory allocated: ", raw_ptr, " (will be freed by defer)");
        // 추가 로직...
        return raw_ptr; // 반환값은 외부에서 수동으로 free해야 함 (defer는 이 함수에서만 동작)
                        // 주의: 이 함수에서 raw_ptr를 반환하면, defer가 호출될 때 이미 유효하지 않은 포인터가 될 수 있음
                        // defer는 함수가 끝날 때까지 raw_ptr가 유효하다고 가정
                        // ===> defer는 반환되는 리소스에는 적용하지 않는 것이 일반적.
                        // 이 예제는 단순히 defer 동작을 보여주기 위한 것.
    }
}


func main() {
    std.println("--- Femto C Defer Example ---");

    // 1. 정상 종료 시 defer 동작 확인
    std.println("\nScenario 1: Normal completion");
    process_data_in_file("output_normal.txt", 0);
    // "File closed." 메시지가 출력될 것임.

    // 2. 예외 발생 시 defer 동작 확인
    std.println("\nScenario 2: Exception during processing");
    try {
        process_data_in_file("output_error.txt", 1);
    } catch (var e: box std.err) {
        std.println("Caught std.err: Code=", e.code, ", Message=", &e.msg);
    }
    // "File closed." 메시지가 예외 처리 후에도 출력될 것임.

    // 3. 메모리 할당 및 defer 해제 (반환값 없는 경우)
    std.println("\nScenario 3: Malloc with defer cleanup (non-returned ptr)");
    func !internal_malloc_example() {
        unsafe {
            var temp_ptr: *void = c.stdlib.malloc(32);
            if (temp_ptr == void) { return; }
            defer c.stdlib.free(temp_ptr); // 이 함수 스코프 벗어날 때 temp_ptr 해제
            std.println("Temp memory allocated: ", temp_ptr, " (will be freed automatically)");
            // temp_ptr 사용...
        }
    }
    !internal_malloc_example(); // 함수 호출 후 temp_ptr 자동 해제
    std.println("After internal_malloc_example, temp memory should be freed.");

    std.println("\n--- End of Example ---");
}
