#include <std>; // 전역 저장소에 있는 표준 라이브러리 포함 (가정)
#include "my_lib"; // 로컬 파일에 정의된 사용자 정의 타입 포함 (가정)

// 사용자 정의 구조체
Coord {
    int x;
    32bit int y; // 32비트 정수 y
}

// 사용자 정의 열거형
LogLevel {
    INFO;
    WARNING;
    ERROR;
}

// _로 시작하는 함수는 export 되지 않습니다. (캡슐화 예시)
void _log_message(LogLevel level, 8bit int* message) {
    if (level == LogLevel.INFO) {
        std.print("info 로그\n");
    } else if (level == LogLevel.WARNING) {
        std.print("warning 로그\n");
    } else if (level == LogLevel.ERROR) {
        std.print("error 로그\n");
    }
    std.println(message);
}

// 두 정수를 더하는 함수
int add(int a, int b) {
    return a + b;
}

// 배열의 모든 요소를 더하는 함수
int sum_array(int[] arr_ptr, int size) { // arr_ptr은 T*
    int total = 0;
    int i = 0;
    while (i < size) {
        total = total + arr_ptr[i]; // *(arr_ptr + i)와 동일
        i = i + 1;
    }
    return total;
}

// 메인 함수 (프로그램 진입점)
int main() {
    // 1. 변수 선언 및 초기화
    int num1 = 10;
    unsigned 32bit int num2 = 20; // 부호 없는 32비트 정수
    float pi = 3.14;
    bool is_active = true;

    // 2. 타입 캐스팅
    float casted_float = (float)num1;
    int casted_int = (int)pi;

    // 3. 배열 선언 및 초기화
    int[5] my_array = {1, 2, 3, 4, 5};
    int[3] another_array;
    another_array[0] = 100;
    another_array[1] = 200;
    another_array[2] = 300;

    // 4. 포인터와 메모리
    int* dynamic_array;
    int array_size = 4;
    dynamic_array = (int*)malloc(sizeof(int) * array_size); // 힙 메모리 할당

    if (dynamic_array != (void*)0) { // void 키워드로 NULL 체크
        dynamic_array[0] = 50;
        dynamic_array[1] = 60;
        dynamic_array[2] = 70;
        dynamic_array[3] = 80;

        // 5. 함수 호출
        int dynamic_sum = sum_array(dynamic_array, array_size);

        // 6. 비트 연산자 (예시: 첫 번째 요소를 2배)
        dynamic_array[0] = dynamic_array[0] << 1;

        // 7. 관계 및 논리 연산자
        if (dynamic_sum > 200 && is_active) {
            _log_message(LogLevel.INFO, "Dynamic array sum is large and active.");
        }
        
        free(dynamic_array); // 힙 메모리 해제
    } else {
        _log_message(LogLevel.ERROR, "Failed to allocate memory.");
    }

    // 8. 구조체 사용
    Coord point;
    point.x = num1;
    point.y = num2;

    // 9. 함수 포인터 (sum_array 함수의 포인터)
    int sum_func_ptr(int[], int); // 함수 선언과 동일
    sum_func_ptr = sum_array;

    int static_sum = sum_func_ptr(my_array, 5); // 함수 포인터를 통해 호출

    // 10. 열거형 사용
    LogLevel current_level = LogLevel.WARNING;
    _log_message(current_level, "This is a warning message.");

    // 11. 산술 연산 및 증감 (증감 연산자가 없으므로 수동)
    num1 = num1 + 1; // num1++ 대신

    // 12. #raw_asm 지시자 예시
    #raw_asm {
        mov rax, 1; // 어셈블리어 코드 삽입 예시
        mov rdi, 1;
        mov rsi, 0x1;
        mov rdx, 4;
        syscall;
    }

    return 0; // 프로그램 종료
}
